# requirements-mlops.txt
# Dependencies for the Self-Healing MLOps Data Pipeline
# Versions use compatible-release ranges (>=min,<next_major) so pip can pick
# a wheel that exists for the current Python version (3.13+) while staying
# within a tested minor series.
#
# Install: pip install -r requirements-mlops.txt

# ── Data Ingestion ────────────────────────────────────────────────────────────
requests>=2.32.0,<3          # HTTP client for Open-Meteo and TomTom APIs
urllib3>=2.2.0,<3            # Retry logic (used by requests via HTTPAdapter)

# ── Data Processing ───────────────────────────────────────────────────────────
numpy>=2.0.0,<3              # numpy 2.x required for Python 3.13 wheels; 1.x has none
pandas>=2.2.0,<3             # DataFrames for feature engineering

# ── Experiment Tracking ───────────────────────────────────────────────────────
mlflow>=2.16.0,<3            # Local experiment tracking (mlruns/ directory)
                             # 2.16+ dropped numpy<2 constraint; no server needed

# ── Utilities ────────────────────────────────────────────────────────────────
